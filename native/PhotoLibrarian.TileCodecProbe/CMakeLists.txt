cmake_minimum_required(VERSION 3.20)
project(PhotoLibrarianTileCodecProbe LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(PhotoLibrarian.TileCodecProbe SHARED TileCodec.cpp)
target_compile_features(PhotoLibrarian.TileCodecProbe PRIVATE cxx_std_17)
target_compile_definitions(PhotoLibrarian.TileCodecProbe PRIVATE TILE_CODEC_EXPORTS NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(PhotoLibrarian.TileCodecProbe PRIVATE windowscodecs ole32 oleaut32)
add_executable(PhotoLibrarian.CodecMemoryProbe CodecMemoryProbe.cpp)
target_compile_features(PhotoLibrarian.CodecMemoryProbe PRIVATE cxx_std_17)
target_compile_definitions(PhotoLibrarian.CodecMemoryProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(PhotoLibrarian.CodecMemoryProbe PRIVATE PhotoLibrarian.TileCodecProbe psapi)
